import yfinance as yf
import pandas as pd
from flask import Flask, jsonify
import threading
import time
import requests
import os

app = Flask(__name__)

# الاعدادات
SYMBOL = "GC=F"  # الذهب
TELEGRAM_TOKEN = os.getenv("TELEGRAM_TOKEN", "")  # تحطه في Render
TELEGRAM_CHAT_ID = os.getenv("TELEGRAM_CHAT_ID", "")

last_signal = {"price": 0, "signal": "انتظار", "rsi": 0, "time": ""}

def get_rsi(prices, period=14):
    delta = prices.diff()
    gain = (delta.where(delta > 0, 0)).rolling(window=period).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(window=period).mean()
    rs = gain / loss
    rsi = 100 - (100 / (1 + rs))
    return rsi

def get_signal():
    try:
        data = yf.download(SYMBOL, period="2d", interval="5m", progress=False)
        if data.empty:
            return None
        close = data['Close']
        # متوسطات
        sma20 = close.rolling(20).mean().iloc[-1]
        sma50 = close.rolling(50).mean().iloc[-1]
        price = float(close.iloc[-1])
        rsi = float(get_rsi(close).iloc[-1])

        if price > sma20 and sma20 > sma50 and rsi < 70:
            signal = "🟢 شراء قوي - BUY"
        elif price < sma20 and sma20 < sma50 and rsi > 30:
            signal = "🔴 بيع قوي - SELL"
        elif rsi > 70:
            signal = "⚠️ تشبع شراء - انتظر بيع"
        elif rsi < 30:
            signal = "⚠️ تشبع بيع - انتظر شراء"
        else:
            signal = "⏸️ انتظار - لا تدخل توا"

        return {"price": round(price, 2), "signal": signal, "rsi": round(rsi, 2), "sma20": round(float(sma20),2), "sma50": round(float(sma50),2)}
    except Exception as e:
        print(e)
        return None

def send_telegram(msg):
    if not TELEGRAM_TOKEN or not TELEGRAM_CHAT_ID:
        return
    try:
        url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage"
        requests.post(url, json={"chat_id": TELEGRAM_CHAT_ID, "text": msg})
    except:
        pass

def loop_signals():
    global last_signal
    while True:
        res = get_signal()
        if res:
            last_signal = res
            last_signal["time"] = time.strftime("%Y-%m-%d %H:%M:%S")
            # لو اشارة قوية ابعت تليجرام
            if "قوي" in res["signal"]:
                send_telegram(f"إشارة ذهب \n{res['signal']}\nالسعر: {res['price']}\nRSI: {res['rsi']}")
        time.sleep(300)  # كل 5 دقايق

@app.route("/")
def home():
    return jsonify(last_signal)

@app.route("/signal")
def signal():
    return jsonify(last_signal)

# شغل اللوب في الخلفية
threading.Thread(target=loop_signals, daemon=True).start()

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.getenv("PORT", 10000)))
